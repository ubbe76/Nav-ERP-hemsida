> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** En statiskt exporterad Next.js-sajt på svenska som får svenska tillverkande företag att förstå Nav|ERP och installera det själva.

**Architecture:** Fristående Next.js 16-projekt i `/home/ubbe/Nav-ERP/hemsida` med `output: "export"`. Manualen, skärmdumparna, färgtokens, kapitellistan och logotypkomponenten synkas vid varje bygge från ERP-repot (`../dev` eller en klon) så att de har en enda källa. Sidinnehållet för funktionsområdena är data i `content/`, sidorna är tunna mallar runt den datan.

**Tech Stack:** Next.js 16.2.10 (App Router), React 19, TypeScript strict, vanlig CSS + CSS Modules, `marked` 18, `lucide-react` 0.469.0, Vitest 2, Playwright, GitHub Actions + GitHub Pages.

**Spec:** `docs/superpowers/specs/2026-10-14-hemsida-design.md`

```ts
// Site-wide constants and URL helpers. BASE_PATH/SITE_URL come from the
// environment so one build works on a custom domain and under
// <user>.github.io/<repo>/ (see next.config.ts and the deploy workflow).
export const SITE = {
  name: "Nav|ERP",
  description:
    "Öppet källkod-ERP för svenska tillverkande företag: artiklar, lager, inköp, försäljning, tillverkning, tid, kalkyl och bokföring i ett system. Självhostat och utan licensavgifter.",
  repoUrl: "https://github.com/ubbe76/Nav-ERP",
  cloneUrl: "https://github.com/ubbe76/Nav-ERP.git",
} as const;

export function basePath(): string {
  return process.env.NEXT_PUBLIC_BASE_PATH ?? "";
}

/** Prefix a root-relative path with the base path. For raw HTML and <img> — next/link adds it itself. */
export function withBase(p: string): string {
  return p.startsWith("/") && !p.startsWith("//") ? `${basePath()}${p}` : p;
}

/** Origin the site is served from, without trailing slash. */
export function siteUrl(): string {
  return (process.env.SITE_URL ?? "http://localhost:3000").replace(/\/+$/, "");
}

export function absoluteUrl(p: string): string {
  return `${siteUrl()}${withBase(p)}`;
}
```

```ts
import { cp, mkdir, readdir, rm, stat } from "node:fs/promises";
import path from "node:path";

// Content shared with the ERP repo: the manual, screenshots, colour tokens,
// the chapter list and the logo. Copied in at build time so each has one
// source of truth; the copies are gitignored.
export interface SyncItem {
  /** Path inside the Nav-ERP repo. */
  from: string;
  /** Path inside hemsida. */
  to: string;
  /** For directories: which file names to copy. Absent means `from` is a single file. */
  match?: RegExp;
}

export const SYNC_ITEMS: SyncItem[] = [
  { from: "docs/manual", to: "content/manual", match: /\.md$/ },
  { from: "docs/assets", to: "public/screenshots", match: /^screenshot-.+\.png$/ },
  { from: "apps/web/styles/tokens.css", to: "styles/tokens.css" },
  { from: "apps/web/lib/manual.ts", to: "lib/upstream/manual.ts" },
  { from: "apps/web/components/ui/Wordmark.tsx", to: "components/upstream/Wordmark.tsx" },
  { from: "apps/web/components/ui/Wordmark.module.css", to: "components/upstream/Wordmark.module.css" },
];

export class SyncError extends Error {}

async function exists(p: string): Promise<boolean> {
  try {
    await stat(p);
    return true;
  } catch {
    return false;
  }
}

export async function syncContent(sourceDir: string, targetDir: string, items: SyncItem[] = SYNC_ITEMS): Promise<string[]> {
  // Validate everything before touching the target, so a missing file never
  // leaves a half-synced tree behind.
  const plan: Array<{ src: string; dest: string }> = [];
  for (const item of items) {
    const src = path.join(sourceDir, item.from);
    if (!(await exists(src))) throw new SyncError(`Saknas i Nav-ERP: ${item.from} (letade i ${sourceDir})`);
    const match = item.match;
    if (match) {
      const files = (await readdir(src)).filter((f) => match.test(f)).sort();
      if (files.length === 0) throw new SyncError(`Inga filer matchar ${match} i ${item.from}`);
      for (const f of files) plan.push({ src: path.join(src, f), dest: path.join(item.to, f) });
    } else {
      plan.push({ src, dest: item.to });
    }
  }

  // Directories are replaced wholesale so a removed chapter or screenshot disappears too.
  for (const item of items) {
    if (item.match) await rm(path.join(targetDir, item.to), { recursive: true, force: true });
  }
  for (const { src, dest } of plan) {
    const out = path.join(targetDir, dest);
    await mkdir(path.dirname(out), { recursive: true });
    await cp(src, out);
  }
  return plan.map((p) => p.dest);
}
```

```ts
import { execFileSync } from "node:child_process";
import { existsSync, mkdtempSync, rmSync } from "node:fs";
import os from "node:os";
import path from "node:path";
import { SyncError, syncContent } from "../lib/sync";

// Source: NAV_ERP_DIR, else the sibling checkout ../dev. Only when neither is
// set up do we shallow-clone the public repo (CI checks the repo out itself
// and sets NAV_ERP_DIR, so private repos work there with a token).
const REPO = "https://github.com/ubbe76/Nav-ERP.git";

async function main() {
  const root = process.cwd();
  const explicit = process.env.NAV_ERP_DIR;
  const configured = explicit ?? path.resolve(root, "..", "dev");
  let source = configured;
  let tmp: string | undefined;

  if (!existsSync(configured)) {
    if (explicit) throw new SyncError(`NAV_ERP_DIR pekar på ${configured}, som inte finns`);
    tmp = mkdtempSync(path.join(os.tmpdir(), "nav-erp-"));
    console.log(`sync: ${configured} saknas — klonar ${REPO}`);
    execFileSync("git", ["clone", "--depth", "1", REPO, tmp], { stdio: "inherit" });
    source = tmp;
  }

  try {
    const copied = await syncContent(source, root);
    console.log(`sync: ${copied.length} filer från ${tmp ? REPO : source}`);
  } finally {
    if (tmp) rmSync(tmp, { recursive: true, force: true });
  }
}

main().catch((err: unknown) => {
  console.error(err instanceof SyncError ? `sync: ${err.message}` : err);
  process.exit(1);
});
```

```ts
function boot(opts: { stored?: string | null; storageThrows?: boolean; prefersDark?: boolean }): string | undefined {
```

```ts
  THEME_STORAGE_KEY
```

```tsx
"use client";

import { Ghost, Moon, Sun, type LucideIcon } from "lucide-react";
import { useEffect, useState } from "react";
import { THEMES, THEME_STORAGE_KEY, isThemeId, type ThemeId } from "@/lib/themes";
import styles from "./ThemeToggle.module.css";

const ICONS: Record<ThemeId, LucideIcon> = { light: Sun, dark: Moon, dracula: Ghost };

export function ThemeToggle() {
  // null until mounted: the boot script chose the theme before hydration and
  // the server can't know which one.
  const [theme, setTheme] = useState<ThemeId | null>(null);

  useEffect(() => {
    const current = document.documentElement.getAttribute("data-theme");
    setTheme(isThemeId(current) ? current : "light");
  }, []);

  function select(next: ThemeId) {
    setTheme(next);
    document.documentElement.setAttribute("data-theme", next);
    try {
      window.localStorage.setItem(THEME_STORAGE_KEY, next);
    } catch {
      // Storage blocked — the choice still holds for this page view.
    }
  }

  return (
    <div className={styles.toggle} role="group" aria-label="Färgtema">
      {THEMES.map((t) => {
        const Icon = ICONS[t.id];
        return (
          <button
            key={t.id}
            type="button"
            className={styles.option}
            aria-pressed={theme === t.id}
            aria-label={t.label}
            title={t.label}
            onClick={() => select(t.id)}
          >
            <Icon size={16} aria-hidden />
          </button>
        );
      })}
    </div>
  );
}
```

```ts
import { readFileSync, readdirSync } from "node:fs";
import path from "node:path";
import { THEMES, type ThemeId } from "./themes";

// Screenshots come from Nav-ERP's docs/assets (synced to public/screenshots/)
// and are named screenshot-<key>.png (light), screenshot-<key>-dark.png and
// screenshot-<key>-dracula.png. Not every key exists in every theme, so each
// theme falls back to the closest variant that does.
export const FALLBACK: Record<ThemeId, ThemeId[]> = {
  light: ["light", "dark", "dracula"],
  dark: ["dark", "dracula", "light"],
  dracula: ["dracula", "dark", "light"],
};

export type Variants = Partial<Record<ThemeId, string>>;

export interface ScreenshotImage {
  src: string;
  themes: ThemeId[];
  width: number;
  height: number;
}

export function parseScreenshotName(file: string): { key: string; theme: ThemeId } | null {
  const m = /^screenshot-(.+?)(?:-(dark|dracula))?\.png$/.exec(file);
  if (!m || !m[1]) return null;
  return { key: m[1], theme: (m[2] as ThemeId | undefined) ?? "light" };
}

export function buildRegistry(files: string[]): Map<string, Variants> {
  const registry = new Map<string, Variants>();
  for (const file of files) {
    const parsed = parseScreenshotName(file);
    if (!parsed) continue;
    const variants = registry.get(parsed.key) ?? {};
    variants[parsed.theme] = file;
    registry.set(parsed.key, variants);
  }
  return registry;
}

/** One entry per distinct file, listing the themes that should show it. */
export function resolveVariants(variants: Variants): Array<{ file: string; themes: ThemeId[] }> {
  const byFile = new Map<string, ThemeId[]>();
  for (const { id } of THEMES) {
    const pick = FALLBACK[id].map((t) => variants[t]).find((f): f is string => !!f);
    if (!pick) continue;
    byFile.set(pick, [...(byFile.get(pick) ?? []), id]);
  }
  return [...byFile].map(([file, themes]) => ({ file, themes }));
}

/** Width/height from a PNG's IHDR chunk, so <img> reserves its space before loading. */
export function pngSize(buf: Buffer): { width: number; height: number } {
  if (buf.length < 24 || buf.toString("ascii", 12, 16) !== "IHDR") throw new Error("Inte en PNG-fil");
  return { width: buf.readUInt32BE(16), height: buf.readUInt32BE(20) };
}

const DEFAULT_DIR = path.join(process.cwd(), "public", "screenshots");

export function loadScreenshot(key: string, dir: string = DEFAULT_DIR): ScreenshotImage[] {
  const variants = buildRegistry(readdirSync(dir)).get(key);
  if (!variants) {
    throw new Error(`Skärmdumpen "${key}" saknas i ${dir} — finns screenshot-${key}*.png i Nav-ERP/docs/assets?`);
  }
  return resolveVariants(variants).map(({ file, themes }) => ({
    src: `/screenshots/${file}`,
    themes,
    ...pngSize(readFileSync(path.join(dir, file))),
  }));
}
```

- [ ] **Step 3: Implementera `lib/github.ts`**

```ts
export interface RepoStats {
  stars: number | null;
  release: { tag: string; url: string } | null;
}

const API = "https://api.github.com/repos/ubbe76/Nav-ERP";

async function getJson(fetchImpl: typeof fetch, url: string, token: string | undefined): Promise<unknown> {
  try {
    const res = await fetchImpl(url, {
      headers: { Accept: "application/vnd.github+json", ...(token ? { Authorization: `Bearer ${token}` } : {}) },
      // Static export refuses uncached fetches; at build time this is fetched once.
      cache: "force-cache",
      signal: AbortSignal.timeout(5000),
    });
    return res.ok ? await res.json() : null;
  } catch {
    return null;
  }
}

// Fetched at build time. Any failure (rate limit, private repo, offline) only
// drops the numbers from the page — the build never fails on GitHub.
export async function fetchRepoStats(
  fetchImpl: typeof fetch = fetch,
  token: string | undefined = process.env.GITHUB_TOKEN,
): Promise<RepoStats> {
  const [repo, release] = await Promise.all([getJson(fetchImpl, API, token), getJson(fetchImpl, `${API}/releases/latest`, token)]);
  const stars = (repo as { stargazers_count?: unknown } | null)?.stargazers_count;
  const r = release as { tag_name?: unknown; html_url?: unknown } | null;
  return {
    stars: typeof stars === "number" ? stars : null,
    release: typeof r?.tag_name === "string" && typeof r.html_url === "string" ? { tag: r.tag_name, url: r.html_url } : null,
  };
}

let cached: Promise<RepoStats> | undefined;

/** Memoised per build worker: header and footer share one request pair. */
export function getRepoStats(): Promise<RepoStats> {
  return (cached ??= fetchRepoStats());
}
```

```ts
import { readFileSync } from "node:fs";
import path from "node:path";
import { Marked, type Tokens } from "marked";
import { withBase } from "./site";
import { MANUAL_CHAPTERS, getChapter, slugifyHeading, type ManualChapter } from "./upstream/manual";

// The chapter list and the heading slug rule come from the app (synced), so
// /hjalp/<kapitel>#<rubrik> means the same thing on the site and in the app.
// The app's own loader reads a path relative to apps/web, so we load from
// content/manual/ ourselves.
export { MANUAL_CHAPTERS, getChapter, type ManualChapter };

export interface TocEntry {
  id: string;
  text: string;
}

export interface RenderedChapter {
  html: string;
  toc: TocEntry[];
}

const MANUAL_DIR = path.join(process.cwd(), "content", "manual");

const attr = (s: string) => s.replace(/&/g, "&amp;").replace(/"/g, "&quot;");
const plainText = (md: string) => md.replace(/\*\*|__|`/g, "");

// Created per call so withBase() reads the environment at render time.
function createMarked(): Marked {
  return new Marked({
    renderer: {
      heading({ tokens, depth, text }) {
        return `<h${depth} id="${slugifyHeading(text)}">${this.parser.parseInline(tokens)}</h${depth}>\n`;
      },
      link({ href, title, tokens }) {
        const titleAttr = title ? ` title="${attr(title)}"` : "";
        return `<a href="${attr(withBase(href))}"${titleAttr}>${this.parser.parseInline(tokens)}</a>`;
      },
    },
  });
}

export function renderManual(markdown: string): RenderedChapter {
  const marked = createMarked();
  const toc = marked
    .lexer(markdown)
    .filter((t): t is Tokens.Heading => t.type === "heading" && t.depth === 2)
    .map((t) => ({ id: slugifyHeading(t.text), text: plainText(t.text) }));
  return { html: marked.parse(markdown, { async: false }) as string, toc };
}

function readChapter(chapter: ManualChapter, dir: string): string {
  const file = path.join(dir, chapter.file);
  try {
    return readFileSync(file, "utf8");
  } catch {
    throw new Error(`Manualkapitlet "${chapter.slug}" saknar fil: ${file} — kör npm run sync`);
  }
}

export function loadChapter(chapter: ManualChapter, dir: string = MANUAL_DIR): RenderedChapter {
  return renderManual(readChapter(chapter, dir));
}

/** Ids of every heading in a chapter — used to validate deep links into the manual. */
export function chapterAnchors(slug: string, dir: string = MANUAL_DIR): Set<string> {
  const chapter = getChapter(slug);
  if (!chapter) throw new Error(`Okänt manualkapitel: ${slug}`);
  return new Set(
    new Marked()
      .lexer(readChapter(chapter, dir))
      .filter((t): t is Tokens.Heading => t.type === "heading")
      .map((t) => slugifyHeading(t.text)),
  );
}
```

```ts
// The cost chain Nav|ERP's calculation uses (manual: System → Kalkyler),
// for the worked example on the site. All markups are percentages.
export interface CalcInput {
  material: number;
  moPct: number;
  labour: number;
  toPct: number;
  subcontract: number;
  loPct: number;
  semiFinished: number;
  affoPct: number;
  generalPct: number;
}

export interface CalcLine {
  label: string;
  amount: number;
  kind: "base" | "markup" | "subtotal";
}

const pct = (n: number) => `${new Intl.NumberFormat("sv-SE").format(n)} %`;

export function calcChain(i: CalcInput): { lines: CalcLine[]; tvk: number; svk: number; price: number } {
  const mo = (i.material * i.moPct) / 100;
  const to = (i.labour * i.toPct) / 100;
  const lo = (i.subcontract * i.loPct) / 100;
  // Semi-finished goods carry no MO/LO markup — it's already inside their own cost.
  const tvk = i.material + mo + i.labour + to + i.subcontract + lo + i.semiFinished;
  const affo = (tvk * i.affoPct) / 100;
  const svk = tvk + affo;
  const general = (svk * i.generalPct) / 100;
  const price = svk + general;

  const lines: CalcLine[] = [
    { label: "Direkt material", amount: i.material, kind: "base" },
    { label: `MO-påslag ${pct(i.moPct)}`, amount: mo, kind: "markup" },
    { label: "Direkt lön", amount: i.labour, kind: "base" },
    { label: `TO-påslag ${pct(i.toPct)}`, amount: to, kind: "markup" },
    { label: "Legokostnad", amount: i.subcontract, kind: "base" },
    { label: `LO-påslag ${pct(i.loPct)}`, amount: lo, kind: "markup" },
    ...(i.semiFinished > 0 ? [{ label: "Halvfabrikat", amount: i.semiFinished, kind: "base" } as const] : []),
    { label: "Tillverkningskostnad", amount: tvk, kind: "subtotal" },
    { label: `Affo-påslag ${pct(i.affoPct)}`, amount: affo, kind: "markup" },
    { label: "Självkostnad", amount: svk, kind: "subtotal" },
    { label: `Generellt påslag ${pct(i.generalPct)}`, amount: general, kind: "markup" },
    { label: "Försäljningspris", amount: price, kind: "subtotal" },
  ];
  return { lines, tvk, svk, price };
}

/** Made-up numbers for the worked example. */
export const EXAMPLE: CalcInput = {
  material: 1000,
  moPct: 10,
  labour: 600,
  toPct: 40,
  subcontract: 200,
  loPct: 5,
  semiFinished: 0,
  affoPct: 8,
  generalPct: 20,
};

export function formatKr(n: number): string {
  return `${new Intl.NumberFormat("sv-SE", { maximumFractionDigits: 0 }).format(n)} kr`;
}
```

```tsx
"use client";

import { useEffect, useRef, type ReactNode } from "react";

// Fades its content in when scrolled into view. The element is only hidden
// once this effect has run and only if it starts below the fold — so the page
// is complete without JS, with reduced motion, and above the fold.
export function Reveal({ children, className }: { children: ReactNode; className?: string }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el || !("IntersectionObserver" in window)) return;
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    if (el.getBoundingClientRect().top < window.innerHeight) return;
    el.dataset.reveal = "pending";
    const io = new IntersectionObserver(
      ([entry]) => {
        if (!entry?.isIntersecting) return;
        el.dataset.reveal = "done";
        io.disconnect();
      },
      { rootMargin: "0px 0px -10% 0px" },
    );
    io.observe(el);
    return () => io.disconnect();
  }, []);

  return (
    <div ref={ref} className={className}>
      {children}
    </div>
  );
}
```

```tsx
"use client";

import { Check, Copy } from "lucide-react";
import { useEffect, useState } from "react";
import styles from "./CodeBlock.module.css";

export function CodeBlock({ code, label }: { code: string; label?: string }) {
  // The copy button only appears once JS runs and the Clipboard API exists.
  const [canCopy, setCanCopy] = useState(false);
  const [copied, setCopied] = useState(false);

  useEffect(() => setCanCopy(!!navigator.clipboard), []);

  async function copy() {
    try {
      await navigator.clipboard.writeText(code);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {
      // Clipboard denied — the text is still selectable.
    }
  }

  return (
    <div className={styles.block}>
      {label && <div className={styles.label}>{label}</div>}
      <pre className={styles.pre}>
        <code>{code}</code>
      </pre>
      {canCopy && (
        <button type="button" className={styles.copy} onClick={copy} aria-label={copied ? "Kopierat" : "Kopiera"}>
          {copied ? <Check size={16} aria-hidden /> : <Copy size={16} aria-hidden />}
        </button>
      )}
    </div>
  );
}
```

```ts
import { existsSync, readFileSync, readdirSync, statSync } from "node:fs";
import path from "node:path";

export interface BrokenLink {
  page: string;
  href: string;
  reason: string;
}

function htmlFiles(dir: string): string[] {
  return readdirSync(dir, { withFileTypes: true }).flatMap((entry) => {
    const p = path.join(dir, entry.name);
    if (entry.isDirectory()) return htmlFiles(p);
    return entry.name.endsWith(".html") ? [p] : [];
  });
}

/** out/hjalp/lager/index.html → /hjalp/lager/ ; out/404.html → /404.html */
export function urlPathFor(outDir: string, file: string): string {
  const rel = path.relative(outDir, file).split(path.sep).join("/");
  if (rel === "index.html") return "/";
  return `/${rel.endsWith("/index.html") ? rel.slice(0, -"index.html".length) : rel}`;
}

/** The output file a URL path resolves to, the way a static host serves it. */
export function fileFor(outDir: string, urlPath: string): string | null {
  const p = decodeURIComponent(urlPath);
  const candidates = p.endsWith("/") ? [`${p}index.html`] : [p, `${p}.html`, `${p}/index.html`];
  for (const candidate of candidates) {
    const file = path.join(outDir, candidate);
    if (existsSync(file) && statSync(file).isFile()) return file;
  }
  return null;
}

const idsIn = (html: string) => new Set([...html.matchAll(/\sid="([^"]+)"/g)].map((m) => m[1] ?? ""));
const refsIn = (html: string) => [...html.matchAll(/\s(?:href|src)="([^"]*)"/g)].map((m) => (m[1] ?? "").replace(/&amp;/g, "&"));

/** Every internal href/src in the export must reach a file, and every #anchor an id on that page. */
export function checkLinks(outDir: string, basePath = ""): BrokenLink[] {
  const idCache = new Map<string, Set<string>>();
  const idsOf = (file: string) => {
    let ids = idCache.get(file);
    if (!ids) {
      ids = idsIn(readFileSync(file, "utf8"));
      idCache.set(file, ids);
    }
    return ids;
  };

  const broken: BrokenLink[] = [];
  for (const file of htmlFiles(outDir).sort()) {
    const page = urlPathFor(outDir, file);
    for (const href of refsIn(readFileSync(file, "utf8"))) {
      if (!href || /^(?:[a-z][a-z0-9+.-]*:|\/\/)/i.test(href)) continue; // external, mailto:, data:
      const url = new URL(href, `http://site${basePath}${page}`);
      if (basePath && url.pathname !== basePath && !url.pathname.startsWith(`${basePath}/`)) {
        broken.push({ page, href, reason: "saknar base path" });
        continue;
      }
      const target = fileFor(outDir, url.pathname.slice(basePath.length) || "/");
      if (!target) {
        broken.push({ page, href, reason: "sidan eller filen finns inte" });
        continue;
      }
      const anchor = decodeURIComponent(url.hash.slice(1));
      if (anchor && target.endsWith(".html") && !idsOf(target).has(anchor)) {
        broken.push({ page, href, reason: `ankaret #${anchor} finns inte` });
      }
    }
  }
  return broken;
}
```

```ts
export type ThemeBlocks = Record<string, Record<string, string>>;

/**
 * Custom properties per theme, read from :root[data-theme="…"] blocks.
 * Theme-independent `:root { … }` blocks are collected under "base".
 */
export function parseThemeBlocks(css: string): ThemeBlocks {
  const blocks: ThemeBlocks = {};
  const clean = css.replace(/\/\*[\s\S]*?\*\//g, "");
  for (const m of clean.matchAll(/([^{}]+)\{([^{}]*)\}/g)) {
    const selector = (m[1] ?? "").trim();
    const themes = [...selector.matchAll(/data-theme="([\w-]+)"/g)].map((t) => t[1] ?? "");
    const keys = themes.length > 0 ? themes : selector === ":root" ? ["base"] : [];
    for (const key of keys) {
      const block = (blocks[key] ??= {});
      for (const d of (m[2] ?? "").matchAll(/(--[\w-]+)\s*:\s*([^;]+);/g)) block[d[1] ?? ""] = (d[2] ?? "").trim();
    }
  }
  return blocks;
}

export function resolveToken(blocks: ThemeBlocks, theme: string, name: string, depth = 0): string {
  const value = blocks[theme]?.[name] ?? blocks.base?.[name];
  if (value === undefined) throw new Error(`${name} saknas i temat ${theme}`);
  const ref = /^var\((--[\w-]+)\)$/.exec(value);
  if (ref?.[1] && depth < 10) return resolveToken(blocks, theme, ref[1], depth + 1);
  return value;
}

function luminance(hex: string): number {
  let h = hex.replace("#", "");
  if (h.length === 3) h = [...h].map((c) => c + c).join("");
  if (!/^[0-9a-f]{6}$/i.test(h)) throw new Error(`Inte en hex-färg: ${hex}`);
  const [r, g, b] = [0, 2, 4].map((i) => {
    const v = parseInt(h.slice(i, i + 2), 16) / 255;
    return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4;
  }) as [number, number, number];
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}

/** WCAG 2 contrast ratio between two hex colours. */
export function contrastRatio(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x) as [number, number];
  return (hi + 0.05) / (lo + 0.05);
}
```

```yaml
      - uses: actions/checkout@v4
        with:
          repository: ubbe76/Nav-ERP
          path: .nav-erp
          token: ${{ secrets.NAV_ERP_TOKEN || github.token }}

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm

      - id: pages
        uses: actions/configure-pages@v5

      - run: npm ci
      - run: npm run typecheck
      - run: npm test

      # e2e runs against a build at the root path (the test server serves out/ at /).
      - run: npm run build
      - run: npx playwright install --with-deps chromium
      - run: npm run test:e2e

      # The published build: origin + base path from GitHub Pages. postbuild
      # re-runs the link check with BASE_PATH, so every link carries the prefix.
      - run: npm run build
        env:
          SITE_URL: ${{ steps.pages.outputs.origin }}
          BASE_PATH: ${{ steps.pages.outputs.base_path }}

      - uses: actions/upload-pages-artifact@v3
        with:
          path: out

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```

````markdown
## Publicering

GitHub Actions (`.github/workflows/deploy.yml`) bygger, testar och publicerar till GitHub Pages vid
push till `main` och varje natt. Första gången:

1. Skapa ett GitHub-repo för hemsidan och pusha.
2. Settings → Pages → Source: **GitHub Actions**.
3. Är Nav-ERP privat: skapa en finkornig token med läsrätt till Nav-ERP och spara den som
   secreten `NAV_ERP_TOKEN`.
4. Egen domän: Settings → Pages → Custom domain. `SITE_URL` och `BASE_PATH` följer med automatiskt.
````
